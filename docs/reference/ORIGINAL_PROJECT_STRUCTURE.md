# Project File Structure

```
alma-dashboard/
├── .env                              # VITE_API_URL=http://localhost:8000
├── index.html
├── package.json
├── tsconfig.json
├── vite.config.ts
├── tailwind.config.ts
│
├── public/
│   └── data/                         # Pre-computed JSON (generated by data-prep scripts)
│       ├── sentences.json            # All sentences (~20K rows, ~15MB)
│       ├── essays.json               # Essay index (essay_id → metadata)
│       ├── unannotated_essays.json   # Unannotated essays with full text
│       ├── splits_stats.json         # Train/val/test statistics
│       ├── baseline_results.json     # All baseline + model results
│       ├── evaluation_results.json   # Detailed per-theme metrics
│       ├── training_history.json     # Epoch-by-epoch training curves
│       ├── group_analysis.json       # Multi-label group results
│       ├── embeddings_umap.json      # Pre-computed UMAP 2D coordinates
│       ├── embeddings_pca.json       # Pre-computed PCA 2D coordinates
│       ├── dataset_versions.json     # V1/V2/V3/V4 summary
│       ├── theme_colors.json         # Canonical color mapping
│       ├── paper_content.json        # Structured content for Paper page
│       └── plots/                    # Static plot images from thesis
│
├── src/
│   ├── main.tsx
│   ├── App.tsx                       # Router + layout
│   ├── index.css                     # Tailwind + globals
│   ├── constants/
│   │   ├── themes.ts                 # THEMES, THEME_COLORS, ABBREVIATIONS
│   │   └── routes.ts
│   ├── types/
│   │   ├── data.ts                   # SentenceRecord, EssayRecord
│   │   ├── inference.ts              # Request/Response types
│   │   └── charts.ts
│   ├── stores/
│   │   ├── dataStore.ts              # Zustand: all data + filters
│   │   ├── inferenceStore.ts         # Zustand: model + results
│   │   └── explorationStore.ts       # Zustand: subsets + charts
│   ├── hooks/
│   │   ├── useDataLoader.ts
│   │   ├── useFilteredData.ts
│   │   ├── useInference.ts
│   │   └── useDebounce.ts
│   ├── lib/
│   │   ├── api.ts                    # Fetch wrapper
│   │   ├── filters.ts                # Filter logic
│   │   ├── csv.ts                    # CSV export
│   │   ├── stats.ts                  # Distribution/co-occurrence computation
│   │   └── sentenceSegmenter.ts      # Client-side segmentation
│   ├── components/                   # See 04_UI_WIREFRAMES.md for full tree
│   └── pages/
│       ├── DataExplorer.tsx
│       ├── Inference.tsx
│       ├── Exploration.tsx
│       └── Paper.tsx
│
├── data-prep/                         # Python scripts for data → JSON
│   ├── requirements.txt
│   ├── prepare_sentences.py
│   ├── prepare_essays.py
│   ├── prepare_unannotated.py
│   ├── prepare_embeddings.py
│   ├── prepare_results.py
│   ├── prepare_training_history.py
│   ├── prepare_paper_content.py
│   ├── validate_data.py
│   └── run_all.sh
│
└── inference-server/                  # Python FastAPI
    ├── requirements.txt
    ├── main.py                       # FastAPI app + routes
    ├── models/
    │   ├── aware_model.py            # Model loading + inference
    │   ├── tfidf_model.py            # TF-IDF baseline
    │   └── sentence_segmenter.py     # spaCy segmentation
    ├── schemas.py                    # Pydantic models
    └── config.py                     # Paths to model weights
```
